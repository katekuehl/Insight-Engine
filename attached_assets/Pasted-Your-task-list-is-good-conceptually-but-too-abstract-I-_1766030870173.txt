  ("Meta Ads Import", load_meta_ads_data),
  ("Email Import", load_email_data),
  ("CRM Import", load_crm_data),
  ("Verification", verify_all_data),
]

for step_name, step_func in steps:
  print(f"▶ {step_name}...")
  result = step_func(...)
  if result["status"] == "FAILED":
    print(f"✗ FAILED: {result['error']}")
    return {"status": "FAILED", "failed_at": step_name}
  else:
    print(f"✓ {step_name} complete")

return {"status": "SUCCESS", "all_steps_completed": True}
